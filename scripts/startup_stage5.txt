// ====================================================================
// Стартовый скрипт для комплексного тестирования Этапа 5
// Проверяет команды: help, mkdir, mv, а также проверку модификации VFS в памяти
// ====================================================================

// --- 1. Тестирование команды HELP ---
// Вывод общего списка команд
help

// Справка по конкретным командам
help mkdir
help mv

// Обработка ошибки команды help
help fake_cmd

// --- 2. Тестирование команды MKDIR (модификация VFS в памяти) ---
// Создание новой папки в текущем каталоге
mkdir my_new_folder
ls

// Создание цепочки вложенных каталогов с флагом -p
mkdir -p archive/2026/autumn
ls archive
ls archive/2026

// Обработка ошибки: создание существующей директории без флага -p
mkdir my_new_folder

// Обработка ошибки: создание вложенной директории без флага -p при отсутствии предка
mkdir no_parent/child

// Обработка ошибки: отсутствие аргументов
mkdir

// --- 3. Тестирование команды MV (перемещение и переименование) ---
// Переименование файла в корне
mv root_file.txt welcome_renamed.txt
ls

// Перемещение файла внутрь созданной директории
mv welcome_renamed.txt my_new_folder
ls my_new_folder

// Чтение перемещенного файла (проверка целостности содержимого)
cat my_new_folder/welcome_renamed.txt

// Перемещение каталога внутрь другого каталога
mv my_new_folder archive/2026
ls archive/2026

// Обработка ошибки: перемещение несуществующего файла
mv non_existent.txt somewhere

// Обработка ошибки: попытка переместить каталог в свой собственный подкаталог
mv archive archive/2026/autumn

// Обработка ошибки: пропущен целевой операнд
mv welcome_renamed.txt

// --- 4. Проверка информации о VFS (изменилось количество узлов в памяти) ---
vfs-info

// Завершение работы
exit
