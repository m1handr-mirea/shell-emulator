// ====================================================================
// Стартовый скрипт для комплексного тестирования Этапа 4
// Проверяет команды: ls, cd, echo, cat, tail, vfs-info, exit
// Включает различные режимы работы, аргументы и обработку ошибок
// ====================================================================

// --- 1. Тестирование команды ECHO ---
// Простой вывод
echo Привет, мир UNIX!

// Вывод с сохранением аргументов в кавычках
echo "Аргумент 1 с пробелами" 'Аргумент 2' 12345

// Пустой echo
echo

// --- 2. Тестирование команды LS ---
// Список файлов в корневом каталоге
ls

// Список файлов в указанной директории
ls home

// Список файлов через путь к конкретному файлу
ls root_file.txt

// Обработка ошибки: несуществующий путь
ls nonexistent_directory

// --- 3. Тестирование команды CD ---
// Переход в директорию home/student/projects/shell_emulator
cd home/student/projects/shell_emulator

// Проверка содержимого текущей директории
ls

// Возврат на уровень выше через ..
cd ..
ls

// Переход по абсолютному пути в корень
cd /
ls

// Обработка ошибки: попытка перехода в несуществующую папку
cd fake_folder

// Обработка ошибки: попытка cd в файл
cd root_file.txt

// Обработка ошибки: слишком много аргументов
cd dir1 dir2

// --- 4. Тестирование команды CAT ---
// Вывод текстового файла из корня
cat root_file.txt

// Вывод файла с логами
cat var/log/service/app.log

// Вывод файла, декодированного из Base64
cat home/student/projects/shell_emulator/blob.dat

// Обработка ошибки: файл не найден
cat missing.txt

// Обработка ошибки: попытка cat каталога
cat home

// Обработка ошибки: отсутствие аргументов
cat

// --- 5. Тестирование команды TAIL ---
// Вывод последних строк лог-файла (по умолчанию)
tail var/log/service/app.log

// Вывод последних N строк с флагом -n
tail -n 1 var/log/service/app.log

// Обработка ошибки: несуществующий файл
tail nowhere.log

// Обработка ошибки: передача директории
tail var

// Обработка ошибки: отсутствие аргументов
tail

// --- 6. Информация о VFS и выход ---
vfs-info
exit
