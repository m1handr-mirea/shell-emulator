// ====================================================================
// Стартовый скрипт для комплексного тестирования этапов 1-3
// Проверяет:
// - Служебную команду vfs-info
// - Заглушки ls и cd с различными аргументами и кавычками
// - Обработку ошибок аргументов команд
// - Синтаксис комментариев C++ (//)
// ====================================================================

// 1. Проверяем получение информации о VFS (имя, sha-256 хеш, число узлов)
vfs-info

// 2. Проверяем обработку ошибки: вызов vfs-info с недопустимым аргументом
vfs-info extra_argument

// 3. Тестирование команды ls без аргументов
ls

// 4. Тестирование команды ls с несколькими аргументами и пробелами в кавычках
ls -la "virtual directory" 'test configs'

// 5. Тестирование перехода cd в относительный и абсолютный путь
cd "var/log"
cd /home/student

// 6. Проверка ошибки команды cd при превышении допустимого числа параметров
cd path1 path2

// 7. Проверка ошибки синтаксиса при незакрытых кавычках
ls "unclosed quote

// 8. Проверка ошибки вызова несуществующей команды
unsupported_command

// 9. Корректное завершение сеанса
exit
